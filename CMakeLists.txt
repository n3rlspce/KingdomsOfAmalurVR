cmake_minimum_required(VERSION 3.20)
project(AmalurVRDiagnostics LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Configure with Visual Studio -A Win32; Amalur is x86")
endif()
add_subdirectory(third_party/minhook)
add_library(mgs5vr_core STATIC third_party/mgs5vr/src/core.cpp third_party/mgs5vr/src/arm_ik.cpp)
target_include_directories(mgs5vr_core PUBLIC third_party/mgs5vr/include)
target_compile_features(mgs5vr_core PUBLIC cxx_std_20)
add_library(amalur_diagnostic SHARED src/diagnostic/proxy.cpp src/diagnostic/exports.def)
target_compile_features(amalur_diagnostic PRIVATE cxx_std_17)
target_compile_options(amalur_diagnostic PRIVATE /W4 /EHsc)
target_link_libraries(amalur_diagnostic PRIVATE minhook mgs5vr_core d3dcompiler dxguid user32)
set_target_properties(amalur_diagnostic PROPERTIES OUTPUT_NAME d3d11)
add_executable(diagnostic_harness src/diagnostic/harness.cpp)
target_compile_features(diagnostic_harness PRIVATE cxx_std_17)
target_compile_options(diagnostic_harness PRIVATE /W4 /EHsc)
target_link_libraries(diagnostic_harness PRIVATE d3d11 dxgi user32)
add_executable(tracking_check src/tracking/tracking_check.cpp)
target_link_libraries(tracking_check PRIVATE mgs5vr_core)

target_compile_features(stereo_check PRIVATE cxx_std_17)
target_link_libraries(stereo_check PRIVATE d3d11 dxgi d3dcompiler)

add_executable(dodge_check src/tracking/dodge_check.cpp)

add_executable(movement_check src/tracking/movement_check.cpp)
